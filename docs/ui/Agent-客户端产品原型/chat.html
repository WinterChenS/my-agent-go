import { request } from './transport' export async function listOrders(page: number, size: number) { // v2 移除了 total 字段，分页改为游标 return request(`/v1/orders?page=${page}&size=${size}`) }

有 3 处会真正断掉，其余是编译期就能发现的重命名：

  • 鉴权头：X-Api-Key 在 v2 被删除，客户端必须改用 Authorization: Bearer。1
  • 分页契约：page/size 换成 cursor/limit，响应体不再返回 total，前端分页器会读到 undefined。2
  • 错误结构：{ error: "..." } 变更为 { error: { code, message } }，现有 catch 分支会拿到对象。3

建议先在 api-client.ts 加一层适配，把游标包装成旧的 page/size 语义，这样 web 端可以独立上线，不依赖后端灰度窗口。
